// ════════════════════════════════════════════════════════════════════════════
// The whole per-edit loop runs here — model → table HTML → arrows — with no R involved.
// Sections: model · formatting · cells and rows · state, baselines and history · editing (as in
// app_xrisk) · arrows and column fitting (as in app_xrisk) · controls.
// ════════════════════════════════════════════════════════════════════════════

// ── model (mirrors app_xrisk's R: START, ALL_RATES, POP0, LIFE_EXP, model_for, fit_state, compute) ──
var START = 2026, LIFE_EXP = 75, POP0 = 8.3e9;
var DEF_EXP = 3;                                  // default horizon 10^3 years; the slider's R default must match
var ALL_RATES = [0.0005, 0.0005, 0.0001, 0.00005, 0.00001];   // per period: 0.05%, 0.05%, 0.01%, 0.005%, 0.001% a year
var COL_RATE = '#e8836f', COL_SURV = '#4fb8d6', COL_POP = '#e46ba6';
var GOOD = '#7fd18f', BAD = '#ee4d4d', ZERO = '#8a97a3';
var ROWS = ['rate', 'surv', 'years', 'gens', 'pop', 'popexp', 'ly', 'lives'];   // display order

// Checkpoints are START plus every power of ten up to the horizon (2026, 2036, 2126, 3026, ...); the
// periods between them are 10, 90, 900, ... years long.
function modelFor(e){
  var cps = [START];
  for(var k = 1; k <= e; k++) cps.push(START + Number('1e' + k));
  var n = cps.length, segStart = [], segEnd = [], segLen = [], defPops = [];
  for(var i = 0; i < n - 1; i++){ segStart.push(cps[i] + 1); segEnd.push(cps[i + 1]); segLen.push(cps[i + 1] - cps[i]); }
  for(var j = 0; j < n; j++) defPops.push(POP0);
  return {n:n, cps:cps, segStart:segStart, segEnd:segEnd, segLen:segLen, defRates:ALL_RATES.slice(0, n - 1), defPops:defPops};
}
// Fit a {rates, pops} state to a model's size: keep the values for the periods both share, fill the rest
// with defaults (a horizon change keeps the reader's edits to the early periods).
function fitState(s, M){
  function fill(x, def){ var out = def.slice(); for(var k = 0; k < Math.min(x.length, def.length); k++) out[k] = +x[k]; return out; }
  return {rates: fill(s.rates, M.defRates), pops: fill(s.pops, M.defPops)};
}
// The cascade, year by year exactly as the R version, read off at the checkpoints. Potential population
// is interpolated geometrically between checkpoints; a checkpoint year takes the value of the period that
// starts there (as build_pop, where the later period overwrites the shared endpoint). Survival is the
// running product of (1 - rate); the cumulative totals count every year after START.
function compute(rates, pops, M){
  var cps = M.cps, n = M.n;
  var o = {rates:rates, growth:[NaN], surv:[], popPot:[], popExp:[], cumYears:[], cumLy:[], cumLyPot:[],
           lives:[], gens:[], livesPot:[]};
  var surv = 1, cy = 0, cly = 0, clyp = 0;
  function record(pop){
    o.surv.push(surv); o.popPot.push(pop); o.popExp.push(pop * surv);
    o.cumYears.push(cy); o.cumLy.push(cly); o.cumLyPot.push(clyp);
    o.lives.push(cly / LIFE_EXP); o.gens.push(cy / LIFE_EXP); o.livesPot.push(clyp / LIFE_EXP);
  }
  record(pops[0]);
  for(var i = 0; i < n - 1; i++){
    var y0 = cps[i], y1 = cps[i + 1], v0 = pops[i], ratio = pops[i + 1] / v0, span = y1 - y0, r = rates[i];
    var pop = v0;
    for(var y = y0 + 1; y <= y1; y++){
      pop = (y === y1 && i < n - 2) ? pops[i + 1] : v0 * Math.pow(ratio, (y - y0) / span);
      surv *= 1 - r;
      cy += surv; cly += pop * surv; clyp += pop;
    }
    record(pop);
    o.growth.push(Math.pow(pops[i + 1] / pops[i], 1 / span) - 1);
  }
  return o;
}

// ── formatting (R's formatC / f_int / group3 equivalents) ──
// thousands separators (a plain space) into a digit string: "1234567" → "1 234 567"
function group3(s){
  var out = '';
  for(var i = 0; i < s.length; i++){ if(i > 0 && (s.length - i) % 3 === 0) out += ' '; out += s.charAt(i); }
  return out;
}
// formatC(x, format = "f", digits = d[, big.mark = " "])
function fmtF(x, d, big){
  if(isNaN(x)) return 'NaN';
  if(!isFinite(x)) return x > 0 ? 'Inf' : '-Inf';
  var s = x.toFixed(d), neg = s.charAt(0) === '-';
  if(neg) s = s.slice(1);
  if(big){ var p = s.split('.'); p[0] = group3(p[0]); s = p.join('.'); }
  return (neg ? '-' : '') + s;
}
function fInt(x){ return fmtF(Math.round(x), 0, true); }
function fPct(x){ return fmtF(x * 100, 4) + '%'; }
function fYears(x){ return fmtF(x, 4, true) + 'y'; }
function fmtYear(y){ return y >= 10000 ? group3(String(y)) : String(y); }   // thin separator only for 5+ digits
// baseline text for the ref slot: same display scale / decimals as the value it sits beside
function fRef(v, toDisp, decimals, suffix){ return '(' + fmtF(v * toDisp, decimals, true) + suffix + ')'; }
function sty(col){ return col == null ? '' : ' style="color:' + col + ';"'; }
// formatC(x, format = "e", digits = 6): R writes at least two exponent digits (1.000000e-05)
function expR(x){ return x.toExponential(6).replace(/e([+-])(\d)$/, function(m, s, d){ return 'e' + s + '0' + d; }); }

// ── cells and rows: the same markup app_xrisk's R builds ──
// Digit slots for a number (value or its delta), as HTML text. See build_slots in app_xrisk for the
// layout rules. o: pad, editable (default true), digitColor(is0, lead), signChar, signColor, refTxt, refGhost.
function buildSlots(nat, toDisp, decimals, suffix, o){
  o = o || {};
  var editable = o.editable !== false, pad = o.pad, digitColor = o.digitColor, k;
  var parts = fmtF(Math.abs(nat) * toDisp, decimals).split('.'), ip = parts[0];
  if(pad != null && ip.length < pad) ip = new Array(pad - ip.length + 1).join('0') + ip;
  ip = group3(ip);
  var chars = (parts.length > 1 ? ip + '.' + parts[1] : ip).split(''), n = chars.length;
  var dot = chars.indexOf('.'); if(dot < 0) dot = n;
  var isdig = chars.map(function(c){ return c >= '0' && c <= '9'; });
  var place = new Array(n), nInt = 0, seenInt = 0, dec = 0;   // place value of each digit, in native units
  for(k = 0; k < dot; k++) if(isdig[k]) nInt++;
  for(k = 0; k < n; k++){
    if(!isdig[k]) continue;
    if(k < dot){ seenInt++; place[k] = Number('1e' + (nInt - seenInt)) / toDisp; }
    else { dec++; place[k] = Number('1e-' + dec) / toDisp; }
  }
  var lead = [], seen = false;                                  // zeros before the first significant digit
  for(k = 0; k < n; k++){
    lead[k] = false;
    if(isdig[k]){ if(!seen && chars[k] === '0') lead[k] = true; else seen = true; }
  }
  var sepCol = digitColor ? digitColor(false, false) : null;
  var up = editable ? '<span class="ar up">&#9650;</span>' : '';
  var dn = editable ? '<span class="ar dn">&#9660;</span>' : '';
  var sp = editable ? '<span class="arsp"></span>' : '';
  function sepSlot(txt, cls, col){
    return '<span class="sep">' + sp + '<b class="' + (cls || 'dch') + '"' + sty(col === undefined ? sepCol : col) + '>' +
           txt + '</b>' + sp + '</span>';
  }
  var groups = [], g = '';                     // thousands-groups: a space starts a new one
  for(k = 0; k < n; k++){
    var ch = chars[k], slot;
    if(isdig[k]){
      var col = digitColor ? digitColor(ch === '0', lead[k]) : ((pad != null && lead[k]) ? ZERO : null);
      slot = '<span class="dig" data-delta="' + expR(place[k]) + '">' + up +
             '<b class="dch"' + sty(col) + '>' + ch + '</b>' + dn + '</span>';
    } else slot = sepSlot(ch === ' ' ? '&nbsp;' : ch);
    if(ch === ' ' && k > 0){ groups.push(g); g = ''; }
    g += slot;
  }
  groups.push(g);
  if(o.signChar != null) groups[0] = sepSlot(o.signChar, 'dch sgn', o.signColor == null ? null : o.signColor) + groups[0];
  if(suffix) groups[groups.length - 1] += sepSlot(suffix, 'dch suf');
  var out = groups.map(function(x){ return '<span class="dgroup">' + x + '</span>'; }).join('');
  if(o.refTxt != null)
    out += '<span class="dgroup">' + sepSlot(o.refTxt, o.refGhost ? 'dch vref ghost' : 'dch vref', null) + '</span>';
  return out;
}
// a value cell on the grid (spans 2 half-columns); node = the arrow endpoint id ("row:j")
function vcellHtml(col, inner, cls, attrs){
  return '<div class="' + (cls || 'vcell') + '" style="grid-column:' + col + '/span 2;"' + (attrs || '') + '>' + inner + '</div>';
}
function nodeAttr(node){ return node ? ' data-node="' + node + '"' : ''; }
// delta colouring shared by both value-cell kinds: grey ± when the change rounds to zero in display
function deltaLook(dv, toDisp, decimals, good, eps){
  var cpos = good === 'more_bad' ? BAD : GOOD, cneg = good === 'more_bad' ? GOOD : BAD;
  var dshow = fmtF(Math.abs(dv) * toDisp, decimals);

// Dependency arrows (see arrow_edges in app_xrisk): for checkpoints j >= 2 each value comes from its
// row's previous value (h) and from rows above; a source in a hidden row is replaced by what that row
// was derived from, and an arrow that skips visible rows is type d.
var CROSS = {
  surv:   function(j){ return ['rate:' + (j - 1)]; },           // cs_t <- ae over the period ending at t
  years:  function(j){ return ['surv:' + j]; },                 // Y_t  <- cs_t
  gens:   function(j){ return ['surv:' + j]; },                 // G_t  <- cs_t
  popexp: function(j){ return ['pop:' + j, 'surv:' + j]; },     // E_t  <- P_t, cs_t
  ly:     function(j){ return ['popexp:' + j]; },               // H_t  <- E_t
  lives:  function(j){ return ['popexp:' + j]; }                // N_t  <- E_t
};
var HORIZ = ['surv', 'years', 'gens', 'ly', 'lives'];            // running totals
function arrowEdges(vis, n){
  function addU(arr, x){ if(arr.indexOf(x) < 0) arr.push(x); }
  function expand(id){
    var p = id.split(':'), row = p[0];
    if(vis.indexOf(row) >= 0) return [id];
    var out = [];
    if(CROSS[row]) CROSS[row](+p[1]).forEach(function(s){ expand(s).forEach(function(x){ addU(out, x); }); });
    return out;
  }
  var out = [];
  vis.forEach(function(row){
    for(var j = 2; j <= n; j++){
      if(HORIZ.indexOf(row) >= 0) addU(out, row + ':' + (j - 1) + ',' + row + ':' + j + ',h');
      var srcs = [];
      if(CROSS[row]) CROSS[row](j).forEach(function(s){ expand(s).forEach(function(x){ addU(srcs, x); }); });
      srcs.forEach(function(s){
        var skips = vis.indexOf(row) - vis.indexOf(s.split(':')[0]) > 1;
        addU(out, s + ',' + row + ':' + j + ',' + (skips ? 'd' : 'v'));
      });
    }
  });
  return out.join(';');
}

function recordHistory(){
  if(histSuppress) return;
  var cur = snap();
  if(histEq(cur, histStack[histIdx])) return;        // no actual change
  histStack = histStack.slice(0, histIdx + 1);         // drop any redo tail
  histStack.push(cur); histIdx = histStack.length - 1;
  updateHistBtns();
}
function restoreHist(){
  var s = histStack[histIdx];
  histSuppress = true; rates = s.rates.slice(); pops = s.pops.slice(); pushState(); histSuppress = false;
  updateHistBtns();
}
function undo(){ if(histIdx > 0){ histIdx--; restoreHist(); } }

// set survival at checkpoint i to target T by solving the interval-i extinction
// rate, backtracing into earlier intervals when T exceeds the prior survival.
function setSurv(i, T){
  T = Math.min(1, Math.max(0, T));               // 100% ceiling, 0 floor
  for(var m = i; m >= 0; m--){
    var Pprev = 1; for(var k = 0; k < m; k++) Pprev *= Math.pow(1 - rates[k], LEN[k]);
    var needed = T / Pprev;                       // desired (1-r_m)^len_m
    if(needed <= 1){ rates[m] = 1 - Math.pow(needed, 1 / LEN[m]); break; }
    rates[m] = 0;                                 // maxed this interval, keep backtracing
  }
  pushState();
}
